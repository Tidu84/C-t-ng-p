cmake_minimum_required(VERSION 3.22.1)
project(coup_native_engines LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB_RECURSE AB_SOURCES CONFIGURE_DEPENDS engines/abjchess/src/*.cpp)
list(FILTER AB_SOURCES EXCLUDE REGEX "/main\\.cpp$")
add_library(coup_abjchess SHARED ab_bridge.cpp ${AB_SOURCES})
target_include_directories(coup_abjchess PRIVATE engines/abjchess/src)
target_compile_definitions(coup_abjchess PRIVATE NDEBUG IS_64BIT NO_PREFETCH ABJCHESS_RELEASE)
target_compile_options(coup_abjchess PRIVATE -O3 -fvisibility=hidden -fno-char8_t)
target_link_options(coup_abjchess PRIVATE -Wl,-z,max-page-size=16384)
add_library(coup_jqe SHARED jqe_bridge.cpp engines/jqe/src/tables.cpp engines/jqe/src/position.cpp engines/jqe/src/eval.cpp engines/jqe/src/nnue.cpp engines/jqe/src/search.cpp)
target_include_directories(coup_jqe PRIVATE engines/jqe/src)
target_compile_definitions(coup_jqe PRIVATE NDEBUG)
target_compile_options(coup_jqe PRIVATE -O3 -fvisibility=hidden)
target_link_options(coup_jqe PRIVATE -Wl,-z,max-page-size=16384)
